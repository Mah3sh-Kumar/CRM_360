// src/components/GlobalSearch.tsx — Ctrl+K command palette over live store data
import { useEffect, useRef, useState } from 'react';
import { Search, Funnel, Building2, Contact2, HandCoins, Users, CheckSquare, CalendarDays, ListChecks } from 'lucide-react';
import { useCRM } from '@/store/crm';
import { activities, tasks, meetings } from '@/data/mock';

interface Item { id?: string; label: string; sub: string }
interface Group { label: string; page: string; icon: React.ElementType; items: Item[] }

export default function GlobalSearch({ open, onClose, onNavigate }: {
  open: boolean; onClose: () => void; onNavigate: (page: string, id?: string) => void;
}) {
  const crm = useCRM();
  const [q, setQ] = useState('');
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => { if (open) { setQ(''); setTimeout(() => inputRef.current?.focus(), 0); } }, [open]);
  if (!open) return null;

  const ql = q.trim().toLowerCase();
  const has = (s: string) => s.toLowerCase().includes(ql);

  const groups: Group[] = [
    { label: 'Leads', page: 'leads', icon: Funnel, items: ql ? crm.leads.filter(l => has(l.name) || has(l.company)).slice(0, 5).map(l => ({ id: l.id, label: l.name, sub: `${l.company} · ${l.status}` })) : [] },
    { label: 'Accounts', page: 'accounts', icon: Building2, items: ql ? crm.accounts.filter(a => has(a.name)).slice(0, 5).map(a => ({ id: a.id, label: a.name, sub: `${a.industry} · ${a.city}` })) : [] },
    { label: 'Customers', page: 'customers', icon: Contact2, items: ql ? crm.customers.filter(c => has(c.name)).slice(0, 5).map(c => ({ id: c.id, label: c.name, sub: `${c.city} · ${c.status}` })) : [] },
    { label: 'Contacts', page: 'contacts', icon: Users, items: ql ? crm.contacts.filter(c => has(c.name) || has(c.account)).slice(0, 5).map(c => ({ label: c.name, sub: `${c.title} · ${c.account}` })) : [] },
    { label: 'Opportunities', page: 'opportunities', icon: HandCoins, items: ql ? crm.opportunities.filter(o => has(o.name) || has(o.account)).slice(0, 5).map(o => ({ id: o.id, label: o.name, sub: `${o.stage} · ${o.code}` })) : [] },
    { label: 'Tasks', page: 'tasks', icon: CheckSquare, items: ql ? tasks.filter(t => has(t.title)).slice(0, 4).map(t => ({ label: t.title, sub: `${t.assignee} · ${t.status}` })) : [] },
    { label: 'Meetings', page: 'meetings', icon: CalendarDays, items: ql ? meetings.filter(m => has(m.title) || has(m.account)).slice(0, 4).map(m => ({ label: m.title, sub: `${m.account} · ${m.date}` })) : [] },
    { label: 'Activities', page: 'activities', icon: ListChecks, items: ql ? activities.filter(a => has(a.subject)).slice(0, 4).map(a => ({ label: a.subject, sub: `${a.type} · ${a.status}` })) : [] },
  ];

  const visible = groups.filter(g => g.items.length > 0);

  return (
    <div className="fixed inset-0 z-50 bg-[#0c231e]/40 backdrop-blur-[2px]" onClick={onClose}>
      <div className="mx-auto mt-[14vh] w-[min(580px,92vw)] overflow-hidden rounded-xl border border-border bg-card shadow-2xl" onClick={e => e.stopPropagation()}>
        <div className="flex items-center gap-2.5 border-b border-border px-4">
          <Search className="size-4 text-muted-foreground" />
          <input ref={inputRef} value={q} onChange={e => setQ(e.target.value)}
            placeholder="Search leads, accounts, customers, opportunities…"
            className="h-12 w-full bg-transparent text-[13.5px] outline-none" />
          <kbd className="kicker rounded border border-border px-1.5 py-0.5 text-muted-foreground">ESC</kbd>
        </div>
        <div className="max-h-[46vh] overflow-y-auto p-2">
          {!ql && <div className="p-6 text-center font-mono2 text-[11px] leading-relaxed text-muted-foreground">Type to search across CRM 360.<br />Press <b>Ctrl K</b> anywhere to reopen.</div>}
          {ql && visible.length === 0 && <div className="p-6 text-center font-mono2 text-[11px] text-muted-foreground">No results for “{q}”.</div>}
          {visible.map(g => (
            <div key={g.label} className="mb-1">
              <div className="kicker px-2 py-1.5 text-muted-foreground">{g.label}</div>
              {g.items.map((it, i) => (
                <button key={i} onClick={() => { onNavigate(g.page, it.id); onClose(); }}
                  className="flex w-full items-center gap-2.5 rounded-md px-2 py-2 text-left transition-colors hover:bg-[#f4f7f2]">
                  <g.icon className="size-3.5 shrink-0 text-muted-foreground" />
                  <span className="truncate text-[12.5px] font-medium">{it.label}</span>
                  <span className="ml-auto truncate pl-3 font-mono2 text-[10.5px] text-muted-foreground">{it.sub}</span>
                </button>
              ))}
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}

import { CheckSquare, CalendarDays, AlertTriangle, MapPin, Funnel, ShieldCheck } from 'lucide-react';
import type { LucideIcon } from 'lucide-react';
import { cn } from '@/lib/utils';
import { useCRM } from '@/store/crm';
import type { NotifKind } from '@/store/crm';

const ICONS: Record<NotifKind, LucideIcon> = {
  task: CheckSquare, meeting: CalendarDays, alert: AlertTriangle,
  gps: MapPin, lead: Funnel, audit: ShieldCheck,
};

export default function NotificationsPanel({ open, onClose }: { open: boolean; onClose: () => void }) {
  const { notifications, markAllRead } = useCRM();
  if (!open) return null;
  return (
    <>
      <div className="fixed inset-0 z-30" onClick={onClose} />
      <div className="fixed right-4 top-16 z-40 w-[360px] overflow-hidden rounded-xl border border-border bg-card shadow-xl">
        <div className="flex items-center justify-between border-b border-border px-4 py-3">
          <div className="font-display text-[14px] font-semibold">Notifications</div>
          <button onClick={markAllRead} className="font-mono2 text-[10.5px] font-medium text-[#007f24] hover:underline">Mark all read</button>
        </div>
        <div className="max-h-[60vh] overflow-y-auto">
          {notifications.length === 0 && <div className="p-6 text-center font-mono2 text-[11px] text-muted-foreground">No notifications.</div>}
          {notifications.map(n => {
            const Icon = ICONS[n.kind];
            return (
              <div key={n.id} className={cn('flex gap-3 border-b border-border/60 px-4 py-3 last:border-0', !n.read && 'bg-[#f2f7f0]')}>

                {!n.read && <span className="ml-auto mt-1.5 size-1.5 shrink-0 rounded-full bg-[#b03a26]" />}
              </div>
            );

        </div>

    </>
  );

// src/pages/details/shared.tsx — small building blocks for detail/360 screens
import type { ReactNode } from 'react';
import { ArrowLeft } from 'lucide-react';
import { StatusBadge } from '@/components/shared';
import type { StatusBadge as _SB, Tone } from '@/components/shared'; // (type re-export kept minimal)
import { cn } from '@/lib/utils';
import type { ReactElement } from 'react';

export function BackBar({ onBack, kicker, title, sub, badges, actions }: {
  onBack: () => void; kicker: string; title: string; sub?: string;
  badges?: ReactElement[]; actions?: ReactNode;
}) {
  return (
    <div className="pb-5 border-b border-border">
      <button onClick={onBack} className="mb-3 inline-flex items-center gap-1.5 font-mono2 text-[11px] text-muted-foreground transition-colors hover:text-foreground">
        <ArrowLeft className="size-3.5" /> Back
      </button>
      <div className="flex items-end justify-between gap-4">
        <div>
          <div className="kicker mb-1.5 text-muted-foreground">{kicker}</div>
          <h1 className="font-display text-[24px] leading-none font-semibold tracking-tight">{title}</h1>
          {sub && <p className="mt-1.5 text-[12.5px] text-muted-foreground">{sub}</p>}
          {badges && <div className="mt-2.5 flex flex-wrap items-center gap-1.5">{badges}</div>}
        </div>
        <div className="flex items-center gap-2">{actions}</div>
      </div>
    </div>
  );
}

export function DetailTabs<T extends string>({ tabs, active, onChange }: {
  tabs: { id: T; label: string; count?: number }[]; active: T; onChange: (t: T) => void;
}) {
  return (
    <div className="flex gap-1 border-b border-border">
      {tabs.map(t => (
        <button key={t.id} onClick={() => onChange(t.id)}
          className={cn('-mb-px border-b-2 px-3 py-2 text-[12.5px] font-medium transition-colors',
            active === t.id ? 'border-[#007f24] text-[#007f24]' : 'border-transparent text-muted-foreground hover:text-foreground')}>
          {t.label}
          {typeof t.count === 'number' && <span className="ml-1.5 font-mono2 text-[10px] text-muted-foreground">{t.count}</span>}
        </button>
      ))}
    </div>
  );
}

export function Field({ label, children }: { label: string; children?: ReactNode }) {
  return (
    <div>
      <div className="kicker text-muted-foreground">{label}</div>
      <div className="mt-1 text-[13px] font-medium">{children ?? <span className="text-muted-foreground">—</span>}</div>
    </div>
  );
}

export function EmptyRow({ text }: { text: string }) {
  return <div className="rounded-lg border border-dashed border-border p-6 text-center font-mono2 text-[10.5px] text-muted-foreground">{text}</div>;
}

export { StatusBadge };
export type { Tone };

// src/pages/details/LeadDetail.tsx
import { useState } from 'react';
import { PageHeader, DataTable, StatusBadge, Mono } from '@/components/shared';
import type { Tone } from '@/components/shared';
import { BackBar, DetailTabs, Field, EmptyRow } from '@/pages/details/shared';
import { useCRM } from '@/store/crm';
import { employees, activities, fmtINR } from '@/data/mock';
import type { LeadStatus, Customer, Account } from '@/data/mock';
import { ArrowRight } from 'lucide-react';
import { cn } from '@/lib/utils';

const stages: LeadStatus[] = ['New', 'Contacted', 'Qualified', 'Proposal Sent', 'Negotiation', 'Won', 'Lost'];
const tone = (s: LeadStatus): Tone =>
  s === 'Won' ? 'ink' : s === 'Lost' ? 'red' : s === 'New' ? 'sage' : s === 'Negotiation' ? 'amber' : s === 'Proposal Sent' ? 'blue' : 'green';

export default function LeadDetail({ id, onBack, onNavigate }: {
  id: string; onBack: () => void; onNavigate: (page: string, id?: string) => void;
}) {
  const crm = useCRM();
  const lead = crm.leads.find(l => l.id === id);
  const [tab, setTab] = useState<'overview' | 'history'>('overview');
  const [showConvert, setShowConvert] = useState(false);
  const [owner, setOwner] = useState(lead?.owner ?? '');
  const [custStatus, setCustStatus] = useState<'Active' | 'Prospect'>('Active');
  const [result, setResult] = useState<{ customer: Customer; account: Account } | null>(null);
  const [done, setDone] = useState(false);

  if (!lead) return <div className="space-y-4"><EmptyRow text="Lead not found." /><button onClick={onBack} className="font-mono2 text-[11px] text-[#007f24]">← Back to Leads</button></div>;

  const conv = crm.conversions.find(c => c.leadId === id);
  const converted = !!conv || done;
  const idx = stages.indexOf(lead.status);
  const history = crm.audit.filter(a => a.record === lead.code);

  return (
    <div className="space-y-5">
      <BackBar onBack={onBack} kicker={`Leads · ${lead.code}`} title={lead.name}
        sub={`${lead.company} · via ${lead.source} · owned by ${lead.owner}`}
        badges={[<StatusBadge key="s" label={lead.status} tone={tone(lead.status)} />, converted && <StatusBadge key="c" label="Converted" tone="ink" />]}
        actions={!converted && (
          <button onClick={() => setShowConvert(true)}
            className="inline-flex h-8 items-center gap-1.5 rounded-md bg-[#007f24] px-3 text-[12.5px] font-semibold text-white transition-colors hover:bg-[#0a6b20]">
            Convert Lead
          </button>
        )} />

      {/* lifecycle strip */}
      <div className="flex items-center gap-1">
        {stages.map((s, i) => (
          <div key={s} className="flex flex-1 items-center gap-1">
            <div className={cn('h-1.5 flex-1 rounded-full', i <= idx ? (lead.status === 'Lost' && i === idx ? 'bg-[#b03a26]' : 'bg-[#007f24]') : 'bg-border')} />
          </div>
        ))}
      </div>
      <div className="flex justify-between font-mono2 text-[9.5px] uppercase tracking-wider text-muted-foreground">
        <span>New</span><span>Won / Lost</span>
      </div>

      {result && (
        <div className="rounded-lg border border-[#bcdcc6] bg-[#e3f0e7] px-4 py-3 text-[12.5px] text-[#0f6b2e]">
          ✓ Lead converted — created <b>{result.customer.code}</b> (Customer) and <b>{result.account.name}</b> (Account).
          <button onClick={() => onNavigate('customers', result.customer.id)} className="ml-3 font-semibold underline">View Customer</button>
          <button onClick={() => onNavigate('accounts', result.account.id)} className="ml-2 font-semibold underline">View Account</button>
        </div>
      )}

      <DetailTabs tabs={[{ id: 'overview', label: 'Overview' }, { id: 'history', label: 'History', count: history.length }]} active={tab} onChange={setTab} />

      {tab === 'overview' && (
        <div className="grid gap-4 lg:grid-cols-3">
          <div className="rounded-lg border border-border bg-card p-4 lg:col-span-2">
            <div className="grid grid-cols-2 gap-4 md:grid-cols-3">
              <Field label="Company">{lead.company}</Field>
              <Field label="Phone"><Mono>{lead.phone}</Mono></Field>
              <Field label="Source">{lead.source}</Field>
              <Field label="Owner">{lead.owner}</Field>
              <Field label="Created On"><Mono>{lead.createdOn}</Mono></Field>
              <Field label="Est. Value"><span className="font-display font-semibold">{fmtINR(lead.value)}</span></Field>
            </div>
          </div>
          <div className="rounded-lg border border-border bg-card p-4">
            <div className="kicker mb-3 text-muted-foreground">Related Activities</div>
            <div className="space-y-2.5">
              {activities.filter(a => a.relatedTo === lead.company).map(a => (
                <div key={a.id} className="flex items-center gap-2 text-[12px]">
                  <StatusBadge label={a.type} tone="sage" />
                  <span className="truncate">{a.subject}</span>
                  <Mono className="ml-auto shrink-0 text-muted-foreground">{a.due.slice(5, 10)}</Mono>
                </div>
              ))}
              {activities.filter(a => a.relatedTo === lead.company).length === 0 && <EmptyRow text="No activities yet." />}
            </div>
          </div>
        </div>
      )}

      {tab === 'history' && (
        history.length ? <DataTable head={['Time', 'User', 'Module', 'Action']} rows={history.map(h => [
          <Mono className="text-muted-foreground">{h.time}</Mono>, h.user, h.module, h.action,
        ])} /> : <EmptyRow text="No history recorded for this lead yet." />
      )}

      {/* Convert dialog */}
      {showConvert && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-[#0c231e]/45 p-4" onClick={() => setShowConvert(false)}>
          <div className="w-full max-w-[420px] rounded-xl border border-border bg-card p-5 shadow-2xl" onClick={e => e.stopPropagation()}>
            <h3 className="font-display text-[17px] font-semibold">Convert Lead</h3>
            <p className="mt-1 text-[12.5px] text-muted-foreground">This will create a Customer, an Account and a conversion history entry. The lead will be marked Won.</p>
            <div className="mt-4 space-y-3">
              <div>
                <label className="kicker text-muted-foreground">Account / Customer Owner</label>
                <select value={owner} onChange={e => setOwner(e.target.value)} className="mt-1.5 h-9 w-full rounded-md border border-input bg-card px-2 text-[13px]">
                  {employees.map(e => <option key={e.id} value={e.name}>{e.name} · {e.designation}</option>)}
                </select>
              </div>
              <div>
                <label className="kicker text-muted-foreground">Customer Status</label>
                <select value={custStatus} onChange={e => setCustStatus(e.target.value as 'Active' | 'Prospect')} className="mt-1.5 h-9 w-full rounded-md border border-input bg-card px-2 text-[13px]">
                  <option>Active</option><option>Prospect</option>
                </select>
              </div>
            </div>
            <div className="mt-5 flex justify-end gap-2">
              <button onClick={() => setShowConvert(false)} className="h-8 rounded-md border border-input px-3 text-[12.5px] font-medium">Cancel</button>
              <button onClick={() => {
                const r = crm.convertLead(id, owner, custStatus);

function Shell() {
  const crm = useCRM();
  const [route, setRoute] = useState<{ page: string; id?: string }>({ page: 'dashboard' });
  const [searchOpen, setSearchOpen] = useState(false);
  const [notifOpen, setNotifOpen] = useState(false);
  const [menuOpen, setMenuOpen] = useState(false);
  const [geoMsg, setGeoMsg] = useState('');
  const [now, setNow] = useState(new Date());

  useEffect(() => {
    const t = setInterval(() => setNow(new Date()), 1000);
    const onKey = (e: KeyboardEvent) => {
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') { e.preventDefault(); setSearchOpen(true); }
      if (e.key === 'Escape') { setSearchOpen(false); setNotifOpen(false); setMenuOpen(false); }
    };
    window.addEventListener('keydown', onKey);
    return () => { clearInterval(t); window.removeEventListener('keydown', onKey); };
  }, []);

  if (!crm.user) return <Login />;
  const user = crm.user;

  const go = (page: string, id?: string) => {
    if (!navAllowed(user.role, page)) return;
    setRoute({ page, id });
    setMenuOpen(false);
  };

  const page = navAllowed(user.role, route.page) ? route.page : 'dashboard';
  const unread = crm.notifications.filter(n => !n.read).length;

  const render = () => {
    switch (page) {
      case 'dashboard': return <Dashboard go={(p: string) => go(p)} />;
      case 'leads': return <Leads onOpen={(id: string) => go('lead', id)} />;
      case 'lead': return route.id ? <LeadDetail id={route.id} onBack={() => go('leads')} onNavigate={go} /> : <Leads onOpen={(id: string) => go('lead', id)} />;
      case 'accounts': return <Accounts onOpen={(id: string) => go('account', id)} />;
      case 'account': return route.id ? <AccountDetail id={route.id} onBack={() => go('accounts')} /> : <Accounts onOpen={(id: string) => go('account', id)} />;
      case 'contacts': return <Contacts />;
      case 'opportunities': return <Opportunities onOpen={(id: string) => go('opportunity', id)} />;
      case 'opportunity': return route.id ? <OpportunityDetail id={route.id} onBack={() => go('opportunities')} /> : <Opportunities onOpen={(id: string) => go('opportunity', id)} />;
      case 'activities': return <Activities />;
      case 'tasks': return <Tasks />;
      case 'meetings': return <Meetings />;
      case 'tracking': return <Tracking />;
      case 'visits': return <Visits />;
      case 'companies': return <Masters tab="companies" />;
      case 'employees': return <Masters tab="employees" />;
      case 'customers': return route.id ? <CustomerDetail id={route.id} onBack={() => go('customers')} /> : <Masters tab="customers" onOpen={(id: string) => go('customer', id)} />;
      case 'reports': return <Reports />;
      case 'audit-logs': return <AuditLogs />;
      default: return <Dashboard go={(p: string) => go(p)} />;
    }
  };

  return (
    <div className="flex min-h-screen bg-background">
      {/* Sidebar */}
      <aside className="sticky top-0 flex h-screen w-[228px] shrink-0 flex-col border-r border-border bg-[#0c231e] text-[#cfe3d6]">
        <div className="flex items-center gap-2.5 px-5 pt-5 pb-6">
          <div className="flex size-8 items-center justify-center rounded-md bg-[#85c093]">
            <Compass className="size-4.5 text-[#0c231e]" strokeWidth={2.4} />
          </div>
          <div>
            <div className="font-display text-[15px] font-semibold tracking-tight text-white leading-none">CRM 360</div>
            <div className="kicker mt-1 text-[#6f9480]">{user.role}</div>
          </div>
        </div>
        <nav className="flex-1 overflow-y-auto px-3 pb-4">
          {NAV.map(g => {
            const items = g.items.filter(it => navAllowed(user.role, it.id));
            if (!items.length) return null;
            return (
              <div key={g.group} className="mb-4">
                <div className="kicker px-2 pb-1.5 text-[#5d8171]">{g.group}</div>
                {items.map(it => {
                  const active = page === it.id || (it.id === 'customers' && page === 'customer')
                    || (it.id === 'leads' && page === 'lead') || (it.id === 'accounts' && page === 'account')
                    || (it.id === 'opportunities' && page === 'opportunity');
                  const Icon = it.icon;
                  return (
                    <button key={it.id} onClick={() => go(it.id)}
                      className={cn(
                        'group mb-0.5 flex w-full items-center gap-2.5 rounded-md px-2 py-[7px] text-[13px] font-medium transition-colors',
                        active ? 'bg-[#85c093] text-[#0c231e]' : 'text-[#a9c4b2] hover:bg-white/5 hover:text-white'
                      )}>
                      <Icon className="size-4 shrink-0" strokeWidth={2} />
                      {user.role === 'Employee' && ['leads', 'accounts', 'contacts', 'opportunities', 'activities', 'tasks', 'meetings'].includes(it.id) ? `My ${it.label}` : it.label}
                      {active && <span className="ml-auto size-1.5 rounded-full bg-[#0c231e]" />}
                    </button>
                  );
                })}
              </div>
            );
          })}
        </nav>
        <div className="border-t border-white/10 p-3">
          <button
            onClick={async () => {
              if (crm.checkIn.checkedIn) { crm.checkOut(); setGeoMsg(''); }
              else { const r = await crm.requestCheckIn(); setGeoMsg(r.message); }
            }}
            className={cn('flex w-full items-center gap-2 rounded-md px-3 py-2 text-[12.5px] font-semibold transition-colors',
              crm.checkIn.checkedIn ? 'bg-[#123a30] text-[#8fd4a8]' : 'bg-[#007f24] text-white hover:bg-[#0a6b20]')}>
            {crm.checkIn.checkedIn ? <LogOut className="size-4" /> : <LogIn className="size-4" />}
            {crm.checkIn.checkedIn ? 'Check-Out' : 'Check-In'}
            {crm.checkIn.checkedIn && crm.checkIn.since &&
              <span className="ml-auto flex items-center gap-1 font-mono2 text-[10px]"><Timer className="size-3" />{fmtDur(now.getTime() - crm.checkIn.since.getTime())}</span>}
          </button>
          {crm.checkIn.checkedIn && crm.checkIn.fix && (
            <div className="mt-1.5 flex items-center gap-1 px-1 font-mono2 text-[9.5px] text-[#6f9480]">
              <MapPin className="size-3" /> GPS active · {crm.checkIn.fix.lat}, {crm.checkIn.fix.lng}
            </div>
          )}
          {geoMsg && !crm.checkIn.checkedIn && <div className="mt-1.5 px-1 font-mono2 text-[9.5px] text-[#c9b458]">{geoMsg}</div>}
        </div>
      </aside>

      {/* Main */}
      <div className="flex min-w-0 flex-1 flex-col">
        <header className="sticky top-0 z-20 flex h-14 items-center gap-4 border-b border-border bg-background/85 px-6 backdrop-blur">
          <button onClick={() => setSearchOpen(true)} className="relative w-full max-w-sm text-left">
            <Search className="absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
            <span className="flex h-9 w-full items-center rounded-md border border-input bg-card pl-9 pr-12 text-[13px] text-muted-foreground">Search leads, accounts, opportunities…</span>
            <kbd className="kicker absolute right-2.5 top-1/2 -translate-y-1/2 rounded border border-border px-1.5 py-0.5 text-muted-foreground">Ctrl K</kbd>
          </button>
          <div className="ml-auto flex items-center gap-3">
            <StatusBadge label={crm.checkIn.checkedIn ? 'Checked-In' : 'Checked-Out'} tone={crm.checkIn.checkedIn ? 'green' : 'gray'} />
            <button onClick={() => setNotifOpen(o => !o)} className="relative flex size-9 items-center justify-center rounded-md border border-input bg-card transition-colors hover:bg-secondary">
              <Bell className="size-4" />
              {unread > 0 && <span className="absolute right-2 top-2 size-1.5 rounded-full bg-[#b03a26]" />}
            </button>
            <div className="relative">
              <button onClick={() => setMenuOpen(o => !o)} className="flex items-center gap-2.5 rounded-md border border-input bg-card py-1.5 pl-1.5 pr-2.5 transition-colors hover:bg-secondary">
                <div className="flex size-6 items-center justify-center rounded bg-[#09352e] font-mono2 text-[10px] font-semibold text-[#a8d8ba]">{initials(user.name)}</div>
                <div className="text-left leading-tight">
                  <div className="text-[12px] font-semibold">{user.name}</div>
                  <div className="kicker text-muted-foreground">{user.role} · {user.designation}</div>
                </div>
                <ChevronDown className="size-3.5 text-muted-foreground" />
              </button>
              {menuOpen && (
                <>
                  <div className="fixed inset-0 z-30" onClick={() => setMenuOpen(false)} />
                  <div className="absolute right-0 top-11 z-40 w-[240px] rounded-lg border border-border bg-card p-1.5 shadow-lg">
                    <div className="px-2.5 py-2">
                      <div className="text-[12.5px] font-semibold">{user.name}</div>
                      <div className="font-mono2 text-[10.5px] text-muted-foreground">{user.email}</div>
                      <div className="font-mono2 text-[10.5px] text-muted-foreground">{user.company} · {user.code}</div>

                    <button onClick={() => crm.logout()}
                      className="flex w-full items-center gap-2 rounded-md px-2.5 py-2 text-[12.5px] font-medium text-[#b03a26] transition-colors hover:bg-[#fbe9e6]">

                    </button>
                  </div>
                </>
              )}
            </div>
          </div>
        </header>
        <main className="flex-1 px-6 py-6">{render()}</main>
      </div>

      <GlobalSearch open={searchOpen} onClose={() => setSearchOpen(false)} onNavigate={go} />
      <NotificationsPanel open={notifOpen} onClose={() => setNotifOpen(false)} />
    </div>
  );
}

export default function App() {

    <CRMProvider>

  );

function locate(): Promise<{ fix: GeoFix; source: 'gps' | 'fallback' }> {
  return new Promise(resolve => {
    if (!('geolocation' in navigator)) return resolve({ fix: FALLBACK_FIX, source: 'fallback' });
    navigator.geolocation.getCurrentPosition(
      pos => resolve({ fix: { lat: +pos.coords.latitude.toFixed(4), lng: +pos.coords.longitude.toFixed(4) }, source: 'gps' }),
      () => resolve({ fix: FALLBACK_FIX, source: 'fallback' }),
      { timeout: 5000, maximumAge: 60000 },
    );
  });
}

    return null;
  }, []);

  const logout = useCallback(() => {
    setUser(null);
    setCheckIn({ checkedIn: false, since: null, fix: null, address: '' });
  }, []);

  const logAudit = useCallback((module: string, action: string, record: string) => {
    const uname = user?.name ?? 'System';
    setAudit(a => [{ id: ++aidRef.current, time: nowTime(), user: uname, module, action, record }, ...a].slice(0, 200));
  }, [user]);

  const notify = useCallback((kind: NotifKind, title: string, body: string) => {
    setNotifications(n => [{ id: ++nidRef.current, kind, title, body, time: 'Just now', read: false }, ...n].slice(0, 50));
  }, []);

  const markAllRead = useCallback(() => {
    setNotifications(ns => ns.map(n => ({ ...n, read: true })));
  }, []);

  const convertLead = useCallback((leadId: string, owner: string, customerStatus: 'Active' | 'Prospect') => {
    const lead = leads.find(l => l.id === leadId);
    if (!lead) return null;
    const at = nowTime();
    const customer: Customer = {
      id: `cu${++idRef.current}`, code: nextCode('CUS', cusRef.current++), name: lead.company,
      industry: '—', owner, phone: lead.phone, email: '', city: '—', status: customerStatus,
      since: new Date().toISOString().slice(0, 10), revenue: 0,
    };
    const account: Account = {
      id: `a${++acctRef.current}`, name: lead.company, industry: '—', owner,
      contacts: 0, openOpps: 0, revenue: 0, city: '—',
    };
    setCustomers(cs => [customer, ...cs]);
    setAccounts(as => [account, ...as]);
    setLeads(ls => ls.map(l => l.id === leadId ? { ...l, status: 'Won' as const } : l));
    setConversions(cv => [{ leadId, customerId: customer.id, accountId: account.id, at, by: user?.name ?? owner }, ...cv]);
    logAudit('Lead', 'Converted → Customer + Account', lead.code);
    notify('lead', 'Lead Converted', `${lead.company} is now a ${customerStatus} customer`);
    return { customer, account };
  }, [leads, user, logAudit, notify]);

  const stageProb: Record<OppStage, number> = { Prospecting: 15, Qualification: 30, Proposal: 50, Negotiation: 70, Won: 100, Lost: 0 };
  const setOppStage = useCallback((oppId: string, stage: OppStage) => {
    setOpportunities(os => os.map(o => o.id === oppId ? { ...o, stage, probability: stageProb[stage] } : o));
    const opp = opportunities.find(o => o.id === oppId);
    if (opp) {
      logAudit('Opportunity', `Stage → ${stage}`, opp.code);
      if (stage === 'Won') notify('lead', 'Deal Won', `${opp.name} · ${opp.code}`);
    }
  }, [opportunities, logAudit, notify]);

  const requestCheckIn = useCallback(async () => {
    if (checkIn.checkedIn) return { ok: false, source: 'fallback' as const, message: 'Already checked in.' };
    const { fix, source } = await locate();
    setCheckIn({ checkedIn: true, since: new Date(), fix, address: `${fix.lat}, ${fix.lng}` });
    logAudit('Tracking', 'Check-In', source === 'gps' ? `GPS ${fix.lat}, ${fix.lng}` : 'GPS unavailable — fallback fix');
    notify('gps', 'Checked-In', `Workday started at ${nowTime()}`);
    return { ok: true, source, message: source === 'gps' ? 'Checked in with GPS fix.' : 'GPS unavailable — checked in with fallback location.' };
  }, [checkIn.checkedIn, logAudit, notify]);

  const checkOut = useCallback(() => {
    if (!checkIn.checkedIn) return;
    const mins = checkIn.since ? Math.round((Date.now() - checkIn.since.getTime()) / 60000) : 0;
    logAudit('Tracking', 'Check-Out', `Worked ${Math.floor(mins / 60)}h ${mins % 60}m`);
    notify('gps', 'Checked-Out', `Workday ended at ${nowTime()}`);
    setCheckIn({ checkedIn: false, since: null, fix: null, address: '' });
  }, [checkIn, logAudit, notify]);

  const recordVisit = useCallback(({ customer, contact, purpose }: { customer: string; contact: string; purpose: string }) => {
    if (!checkIn.checkedIn) {
      return { ok: false, message: 'Check-In required — you must check in before recording a customer visit.' };
    }
    const visit: Visit = {
      id: `v${++idRef.current}`, employee: user?.name ?? '—', customer,
      checkIn: nowTime(), checkOut: '—', durationMin: 0, purpose,
      outcome: contact ? `With ${contact}` : 'In progress', distanceKm: 0,
    };
    setVisits(vs => [visit, ...vs]);
    logAudit('Visit', 'Started', customer);
    return { ok: true, message: `Visit with ${customer} started at ${visit.checkIn}.` };
  }, [checkIn.checkedIn, user, logAudit]);

  const value = useMemo<CRMStore>(() => ({
    user, login, logout,
    leads, accounts, customers, contacts: seedContacts, opportunities, visits, conversions,
    audit, logAudit, notifications, notify, markAllRead,
    convertLead, setOppStage, checkIn, requestCheckIn, checkOut, recordVisit,
  }), [user, login, logout, leads, accounts, customers, opportunities, visits, conversions,
      audit, logAudit, notifications, notify, markAllRead, convertLead, setOppStage,
      checkIn, requestCheckIn, checkOut, recordVisit]);

  return <CRMContext.Provider value={value}>{children}</CRMContext.Provider>;
}

export function useCRM(): CRMStore {
  const ctx = useContext(CRMContext);
  if (!ctx) throw new Error('useCRM must be used within CRMProvider');
  return ctx;
}

// ---- RBAC (SRS access matrix) ----
export function navAllowed(role: Role, page: string): boolean {
  const adminOnly = ['companies', 'audit-logs'];
  const noEmployee = ['companies', 'employees', 'customers', 'audit-logs', 'tracking'];
  if (role === 'Super Admin' || role === 'Company Admin') return true;
  if (role === 'Manager') return !adminOnly.includes(page);
  return !noEmployee.includes(page);
}
